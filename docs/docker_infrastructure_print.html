3 · DonchianBot — Container principal

Dockerfile

FROM --platform=linux/arm64 python:3.11.9-slim

RUN apt-get update && apt-get install -y --no-install-recommends gcc && \
    rm -rf /var/lib/apt/lists/*

WORKDIR /app
COPY app/requirements.txt .
RUN pip install --no-cache-dir -r requirements.txt

COPY app/ .

RUN useradd -r -u 1000 botuser && chown -R botuser /app
USER botuser

CMD ["python", "-u", "main.py"]
⚠️ Architecture ARM64
--platform=linux/arm64 cible le processeur du Raspi. Toujours builder sur le Raspi via docker compose up -d --build — ne pas builder depuis un Mac Intel.

docker-compose.yml

services:
  donchianbot:
    container_name: donchianbot
    build: .
    restart: unless-stopped
    env_file:
      - /srv/data/donchianbot/.env       # clés API hors du code
    volumes:
      - /srv/data/donchianbot:/app/data  # seul dossier persistant
    environment:
      - DATA_DIR=/app/data
    logging:
      driver: "json-file"
      options:
        max-size: "10m"
        max-file: "3"
⚡ Règle critique de déploiement
❌ JAMAIS
docker compose restart — recharge le container mais pas le code
❌ JAMAIS
docker compose up -d — utilise l'ancienne image sans rebuilder
✅ TOUJOURS
docker compose up -d --build — rebuild l'image ET redémarre

5 · Procédure de duplication sur un nouveau Raspi

1
Sauvegarder le .env depuis l'ancien Raspi
scp raspi-home:/srv/data/donchianbot/.env ~/Desktop/.env.backup
2
Installer Docker sur le nouveau Raspi
curl -fsSL https://get.docker.com | sh
sudo usermod -aG docker $USER
# Se déconnecter / reconnecter pour appliquer
3
Créer la structure des volumes
sudo mkdir -p /srv/data/donchianbot
sudo mkdir -p /srv/data/caddy/{config,data}
sudo mkdir -p /srv/data/homepage/logs
sudo mkdir -p /srv/data/swingbot
sudo chown -R $USER:$USER /srv/data
4
Transférer les données persistantes
# Depuis le Mac (remplacer NOUVEAU_IP par l'IP du nouveau Raspi)
ssh raspi-home "tar -czf - /srv/data/donchianbot" \
  | ssh zorion@NOUVEAU_IP "tar -xzf - -C /"
5
Cloner le code et builder
git clone https://github.com/Zorion83/DonchianBot.git ~/DonchianBot
cd ~/DonchianBot
docker compose up -d --build
6
Vérifier
docker ps -a
docker logs donchianbot --tail=20
